# EBDESIGN Complete Platform Architecture
## Multi-Layer Integrated System for Northeast India Agricultural Ecosystem

**Version:** 2.0 (Complete Platform Specification)  
**Date:** 6 September 2026  
**Scope:** All 18+ interconnected layers for farmer empowerment

---

## PLATFORM OVERVIEW

EBDESIGN is a **complete agricultural operating system** connecting farmers to premium markets through an integrated ecosystem of production, preservation, logistics, knowledge, financing, and compliance services.

**Core Value Proposition:** Farmers produce → digital marketplace finds premium buyers → cold storage preserves quality → logistics delivers → knowledge/subsidy/credit enables scaling → compliance/insurance manages risk.

---

## DATABASE SCHEMA (ALL 24 LAYERS)

```sql
-- CORE TABLES (All interconnected)

-- LAYER 1: FARMERS
CREATE TABLE farmers (
  farmer_id SERIAL PRIMARY KEY,
  name VARCHAR NOT NULL,
  phone VARCHAR UNIQUE,
  state ENUM('assam', 'manipur', 'meghalaya', 'nagaland', 'tripura', 'mizoram', 'sikkim', 'arunachal'),
  district VARCHAR,
  village VARCHAR,
  farm_size_hectares DECIMAL,
  crops_growing VARCHAR[],
  fpo_id INTEGER REFERENCES farmer_producer_organizations(id),
  kyc_verified BOOLEAN,
  bank_account VARCHAR,
  aadhar_id VARCHAR,
  joined_date DATE,
  created_at TIMESTAMP
);

-- LAYER 2: E-COMMERCE (Marketplace)
CREATE TABLE marketplace_products (
  product_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  product_name VARCHAR,
  commodity_type VARCHAR, -- Bhut Jolokia, Kachai Lemon, etc
  quality_grade VARCHAR, -- Grade 1, 2, 3
  quantity_kg INT,
  price_per_kg DECIMAL,
  market_price_reference DECIMAL, -- Dynamically fetched
  listed_date DATE,
  buyer_matches INT,
  orders_received INT,
  repeat_buyer_rate DECIMAL, -- %
  created_at TIMESTAMP
);

-- LAYER 3: COLD STORAGE
CREATE TABLE cold_storage_nodes (
  node_id SERIAL PRIMARY KEY,
  node_name VARCHAR,
  state VARCHAR,
  capacity_tonnes INT,
  current_occupancy_kg INT,
  temp_setpoint DECIMAL,
  energy_consumption_kwh DECIMAL,
  cost_per_kg_per_day DECIMAL,
  created_at TIMESTAMP
);

-- LAYER 4: LABS (Soil, Water, Animal, Food)
CREATE TABLE laboratory_results (
  result_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  test_type ENUM('soil', 'water', 'animal', 'food'),
  test_date DATE,
  results JSONB, -- Flexible schema for different tests
  recommendations TEXT,
  follow_up_action VARCHAR,
  created_at TIMESTAMP
);

-- LAYER 5: DOCTORS (Crop, Vet, Poultry, Fish)
CREATE TABLE doctor_consultations (
  consultation_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  doctor_type ENUM('crop', 'veterinarian', 'poultry', 'fish'),
  doctor_id INTEGER,
  consultation_date DATE,
  problem_description TEXT,
  diagnosis TEXT,
  treatment_plan TEXT,
  follow_up_date DATE,
  cost DECIMAL,
  status ENUM('scheduled', 'completed', 'follow_up'),
  created_at TIMESTAMP
);

-- LAYER 6: EQUIPMENT & SUBSIDY
CREATE TABLE equipment_subsidy (
  subsidy_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  equipment_name VARCHAR,
  government_scheme VARCHAR, -- Pradhan Mantri, State, etc
  eligibility_status VARCHAR, -- eligible, pending, rejected
  subsidy_amount DECIMAL,
  farmer_contribution DECIMAL,
  application_date DATE,
  approval_date DATE,
  disbursement_date DATE,
  status ENUM('applied', 'approved', 'disbursed', 'rejected'),
  created_at TIMESTAMP
);

-- LAYER 7: GST & COMPLIANCE
CREATE TABLE compliance_tracking (
  compliance_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  compliance_type ENUM('gst', 'organic', 'gi', 'fssai'),
  registration_number VARCHAR,
  status ENUM('pending', 'approved', 'active', 'expired'),
  filing_due_date DATE,
  filed_date DATE,
  next_renewal_date DATE,
  documents_uploaded INT,
  created_at TIMESTAMP
);

-- LAYER 8: FINANCE & ACCOUNTING
CREATE TABLE farm_accounting (
  account_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  fiscal_year VARCHAR, -- 2025-26
  income_total DECIMAL,
  expense_total DECIMAL,
  profit_margin DECIMAL,
  cost_per_kg_produced DECIMAL,
  tax_liability DECIMAL,
  created_at TIMESTAMP
);

-- LAYER 9: WEATHER & ALERTS
CREATE TABLE weather_alerts (
  alert_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  alert_type ENUM('frost', 'rainfall', 'drought', 'pest', 'disease', 'market'),
  severity ENUM('low', 'medium', 'high'),
  alert_message TEXT,
  recommended_action TEXT,
  alert_date DATE,
  action_taken BOOLEAN,
  created_at TIMESTAMP
);

-- LAYER 10: PAYMENTS & SETTLEMENT
CREATE TABLE farmer_payments (
  payment_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  order_id INTEGER REFERENCES marketplace_products(product_id),
  buyer_price DECIMAL,
  logistics_cost DECIMAL,
  storage_cost DECIMAL,
  platform_fee DECIMAL,
  farmer_receives DECIMAL,
  payment_method ENUM('upi', 'bank_transfer', 'wallet'),
  payment_date DATE,
  status ENUM('pending', 'processed', 'delivered'),
  created_at TIMESTAMP
);

-- LAYER 11: INSURANCE
CREATE TABLE farmer_insurance (
  insurance_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  insurance_type ENUM('crop', 'livestock', 'accident', 'health'),
  policy_number VARCHAR,
  premium_amount DECIMAL,
  coverage_amount DECIMAL,
  policy_start_date DATE,
  policy_end_date DATE,
  claims_filed INT,
  status ENUM('active', 'expired', 'claimed'),
  created_at TIMESTAMP
);

-- LAYER 12: CREDIT & BANKING
CREATE TABLE farmer_credit (
  credit_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  credit_type ENUM('crop_credit', 'equipment_finance', 'working_capital'),
  amount_sanctioned DECIMAL,
  amount_disbursed DECIMAL,
  interest_rate DECIMAL,
  repayment_period_months INT,
  disbursement_date DATE,
  repayment_schedule JSONB,
  status ENUM('sanctioned', 'disbursed', 'repaying', 'closed'),
  created_at TIMESTAMP
);

-- LAYER 13: CONTRACTS
CREATE TABLE preseason_contracts (
  contract_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  buyer_id INTEGER REFERENCES buyers(buyer_id),
  commodity VARCHAR,
  quantity_kg INT,
  price_per_kg DECIMAL,
  quality_spec TEXT,
  payment_terms VARCHAR,
  contract_start_date DATE,
  contract_end_date DATE,
  input_supply BOOLEAN,
  input_cost DECIMAL,
  status ENUM('drafted', 'signed', 'active', 'completed'),
  dispute_resolution VARCHAR,
  created_at TIMESTAMP
);

-- LAYER 14: FPO/PROJECTS
CREATE TABLE farmer_producer_organizations (
  fpo_id SERIAL PRIMARY KEY,
  fpo_name VARCHAR,
  state VARCHAR,
  members_count INT,
  total_production_capacity_tonnes INT,
  registration_date DATE,
  legal_status ENUM('registered', 'pending', 'inactive'),
  created_at TIMESTAMP
);

-- LAYER 15: BUYERS
CREATE TABLE buyers (
  buyer_id SERIAL PRIMARY KEY,
  buyer_name VARCHAR,
  buyer_type ENUM('specialty_retail', 'chef', 'institutional', 'digital'),
  contact_email VARCHAR,
  phone VARCHAR,
  location VARCHAR,
  product_interest VARCHAR[],
  repeat_orders_count INT,
  average_order_size_kg INT,
  willingness_to_pay DECIMAL, -- per kg
  created_at TIMESTAMP
);

-- LAYER 16: ORDERS & SALES
CREATE TABLE marketplace_orders (
  order_id SERIAL PRIMARY KEY,
  product_id INTEGER REFERENCES marketplace_products(product_id),
  buyer_id INTEGER REFERENCES buyers(buyer_id),
  quantity_kg INT,
  order_date DATE,
  delivery_date DATE,
  quality_on_delivery VARCHAR, -- Grade achieved
  buyer_satisfaction INT, -- 1-5
  repeat_order BOOLEAN,
  order_value DECIMAL,
  created_at TIMESTAMP
);

-- LAYER 17: KNOWLEDGE & LEARNING
CREATE TABLE knowledge_resources (
  resource_id SERIAL PRIMARY KEY,
  resource_type ENUM('video', 'article', 'webinar', 'training'),
  resource_title VARCHAR,
  language VARCHAR, -- English, Hindi, Regional, etc
  topic VARCHAR, -- Crop, Pest, Market, Tech, etc
  url VARCHAR,
  view_count INT,
  download_count INT,
  rating DECIMAL,
  created_at TIMESTAMP
);

-- LAYER 18: MRV (Measurement, Reporting, Verification)
CREATE TABLE mrv_indicators (
  indicator_id SERIAL PRIMARY KEY,
  farmer_id INTEGER REFERENCES farmers(farmer_id),
  indicator_type ENUM('food_loss', 'income', 'food_security', 'nature', 'energy'),
  baseline_value DECIMAL,
  current_value DECIMAL,
  target_value DECIMAL,
  measurement_date DATE,
  verification_status VARCHAR,
  created_at TIMESTAMP
);

-- Interconnecting Views & APIs...
```
