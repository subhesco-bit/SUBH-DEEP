 368: async function generateValueRecommendations(userId, limit = 10) {
 369:   try {
 370:     const preferences = await getConsumerValuePreferences(userId);
 371: 
 372:     // Get products with value scores that match preferences
 373:     const result = await pool.query(
 374:       `SELECT pvs.*, p.name, p.base_price, pvp.final_price, pvp.premium_percentage
 375:        FROM product_value_scores pvs
 376:        LEFT JOIN products p ON pvs.product_id = p.id
 377:        LEFT JOIN product_value_pricing pvp ON pvs.id = pvp.value_score_id
 378:        WHERE pvs.overall_value_score >= COALESCE($1, 0)
 379:        AND (pvs.valid_until IS NULL OR pvs.valid_until > CURRENT_TIMESTAMP)
 380:        ORDER BY pvs.overall_value_score DESC
 381:        LIMIT $2`,
 382:       [preferences.min_value_score, limit]
 383:     );
 384: 
 385:     const recommendations = result.rows.map(row => {
 386:       const matchScore = row.overall_value_score;
 387:       const priceValueRatio = row.final_price > 0 ? matchScore / row.final_price * 100 : 0;
 388: 
 389:       return {
 390:         product_id: row.product_id,
 391:         product_name: row.name,
 392:         value_score: row.overall_value_score,
 393:         value_grade: row.value_grade,
 394:         base_price: row.base_price,
 395:         final_price: row.final_price,
 396:         premium_percentage: row.premium_percentage,
 397:         recommendation_score: matchScore,
 398:         value_match_score: matchScore,
 399:         price_value_ratio: priceValueRatio,
 400:         recommendation_reasons: [
 401:           `High value grade: ${row.value_grade}`,
 402:           `Overall score: ${row.overall_value_score}`
 403:         ]
 404:       };
 405:     });
 406: 
 407:     // Save recommendations
 408:     // DB (M7, fixed 2026-08-16): previously each row was written with its own
 409:     // round trip against the shared pool and no transaction, so a mid-loop
 410:     // failure (e.g. a dropped connection on row 8 of 20) left the user with a
 411:     // partially-updated recommendation set and no rollback. Match the
 412:     // BEGIN/COMMIT/ROLLBACK-on-one-checked-out-client pattern already used by
 413:     // orderService.js's checkout transaction and financialService.js's EMI
 414:     // generation/payment transactions.
 415:     if (recommendations.length > 0) {
 416:       const client = await pool.connect();
 417:       try {
 418:         await client.query('BEGIN');
 419: 
 420:         for (const rec of recommendations) {
 421:           await client.query(
 422:             `INSERT INTO value_recommendations
 423:              (user_id, product_id, recommendation_score, recommendation_reasons,
 424:               value_match_score, price_value_ratio)
 425:              VALUES ($1, $2, $3, $4, $5, $6)
 426:              ON CONFLICT (user_id, product_id)
 427:              DO UPDATE SET
 428:                recommendation_score = EXCLUDED.recommendation_score,
 429:                generated_at = CURRENT_TIMESTAMP`,
 430:             [
 431:               userId,
 432:               rec.product_id,
 433:               rec.recommendation_score,
 434:               JSON.stringify(rec.recommendation_reasons),
 435:               rec.value_match_score,
 436:               rec.price_value_ratio
 437:             ]
 438:           );
 439:         }
 440: 
 441:         await client.query('COMMIT');
 442:       } catch (txErr) {
 443:         await client.query('ROLLBACK');
 444:         throw txErr;
 445:       } finally {
 446:         client.release();
 447:       }
 448:     }
 449: 
 450:     return recommendations;
 451:   } catch (error) {
 452:     logger.error('Generate value recommendations error', { error: error.message, stack: error.stack });
 453:     throw error;
 454:   }
 455: }

 174: /**
 175:  * Translate text from source to target language
 176:  * Uses translation memory first, then external API
 177:  */
 178: async function translateText(sourceText, sourceLang, targetLang, options = {}) {
 179:   const startTime = Date.now();
 180:   
 181:   try {
 182:     // Get language IDs
 183:     const sourceLangResult = await pool.query(
 184:       'SELECT id FROM languages WHERE iso_code = $1',
 185:       [sourceLang]
 186:     );
 187:     const targetLangResult = await pool.query(
 188:       'SELECT id FROM languages WHERE iso_code = $1',
 189:       [targetLang]
 190:     );
 191:     
 192:     if (sourceLangResult.rows.length === 0 || targetLangResult.rows.length === 0) {
 193:       throw new Error('Invalid language code');
 194:     }
 195:     
 196:     const sourceLangId = sourceLangResult.rows[0].id;
 197:     const targetLangId = targetLangResult.rows[0].id;
 198:     
 199:     // Check translation memory first
 200:     const memoryResult = await pool.query(
 201:       'SELECT get_translation_from_memory($1, $2, $3, $4) as memory_id',
 202:       [sourceText, sourceLangId, targetLangId, options.domain || null]
 203:     );
 204:     
 205:     const memoryId = memoryResult.rows[0].memory_id;
 206:     
 207:     let translatedText;
 208:     let usedMemory = false;
 209:     let autoTranslated = false;
 210:     let confidence = 1.0;
 211:     
 212:     if (memoryId) {
 213:       // Get translation from memory
 214:       const memResult = await pool.query(
 215:         'SELECT target_text, confidence_score FROM translation_memory WHERE id = $1',
 216:         [memoryId]
 217:       );
 218:       translatedText = memResult.rows[0].target_text;
 219:       confidence = memResult.rows[0].confidence_score;
 220:       usedMemory = true;
 221:     } else {
 222:       // Use external translation service (Google Cloud Translation)
 223:       translatedText = await translateWithExternalAPI(sourceText, sourceLang, targetLang);
 224:       autoTranslated = true;
 225:       confidence = 0.90; // Default confidence for API translation
 226:       
 227:       // Store in translation memory
 228:       await pool.query(
 229:         `INSERT INTO translation_memory 
 230:          (source_text, source_language_id, target_language_id, target_text, context, domain, confidence_score, is_auto_translated)
 231:          VALUES ($1, $2, $3, $4, $5, $6, $7, $8)`,
 232:         [sourceText, sourceLangId, targetLangId, translatedText, options.context || null, options.domain || null, confidence, true]
 233:       );
 234:     }
 235:     
 236:     const processingTime = Date.now() - startTime;
 237:     
 238:     // Log translation request
 239:     if (options.userId) {
 240:       await pool.query(
 241:         `INSERT INTO translation_requests 
 242:          (user_id, source_language_id, target_language_id, source_text, translated_text, status, used_memory, memory_match_id, processing_time_ms, completed_at)
 243:          VALUES ($1, $2, $3, $4, $5, 'completed', $6, $7, $8, NOW())`,
 244:         [options.userId, sourceLangId, targetLangId, sourceText, translatedText, usedMemory, memoryId, processingTime]
 245:       );
 246:     }
 247:     
 248:     return {
 249:       source_text: sourceText,
 250:       source_language: sourceLang,
 251:       target_language: targetLang,
 252:       translated_text: translatedText,
 253:       confidence: confidence,
 254:       used_memory: usedMemory,
 255:       auto_translated: autoTranslated,
 256:       processing_time_ms: processingTime
 257:     };
 258:   } catch (error) {
 259:     logger.error('Translation error', { error: error.message, stack: error.stack });
 260:     throw error;
 261:   }
 262: }
